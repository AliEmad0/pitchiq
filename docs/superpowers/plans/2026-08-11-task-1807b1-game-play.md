```ts
// src/features/game/view/match-stream.ts
import {
  type DecisionAnswer,
  type MatchDecision,
  defaultAnswer,
} from "@/features/game/domain/match-decisions";
import type { MatchEvent, MatchResult, Side } from "@/features/game/domain/match-types";

export type StreamStep =
  | { kind: "decision"; decision: MatchDecision; events: MatchEvent[] }
  | { kind: "done"; result: MatchResult; events: MatchEvent[] };

/**
 * Drive the interruptible engine, surfacing only the coach's own decisions.
 *
 * ⚠️ EVERY decision the engine raises must be answered or the generator hangs — the
 * opponent's are answered immediately with `defaultAnswer` so it behaves exactly as it
 * does in a batch match. That is a filter on the driver, never a change to the engine.
 *
 * ⚠️ Each step returns only the events that are NEW since the last step. The engine's
 * snapshots are cumulative, so passing them straight through would double-render
 * everything that came before.
 */
export function createStream(
  gen: Generator<MatchDecision, MatchResult, DecisionAnswer>,
  coachSide: Side,
) {
  let seen = 0;
  let finished = false;

  const drain = (
    step: IteratorResult<MatchDecision, MatchResult>,
  ): { step: IteratorResult<MatchDecision, MatchResult>; events: MatchEvent[] } => {
    let current = step;
    // Answer the opponent's decisions without surfacing them.
    while (!current.done && current.value.side !== coachSide) {
      current = gen.next(defaultAnswer(current.value));
    }
    const all = current.done ? current.value.events : current.value.events;
    const fresh = all.slice(seen);
    seen = all.length;
    return { step: current, events: fresh };
  };

  const wrap = (
    step: IteratorResult<MatchDecision, MatchResult>,
    events: MatchEvent[],
  ): StreamStep => {
    if (step.done) {
      finished = true;
      return { kind: "done", result: step.value, events };
    }
    return { kind: "decision", decision: step.value, events };
  };

  return {
    /** Run to the first decision the coach must answer, or to full time. */
    advance(): StreamStep {
      const { step, events } = drain(gen.next(undefined as unknown as DecisionAnswer));
      return wrap(step, events);
    },
    /** Answer the outstanding decision and run on to the next one. */
    answer(a: DecisionAnswer): StreamStep {
      if (finished) throw new Error("match already finished");
      const { step, events } = drain(gen.next(a));
      return wrap(step, events);
    },
  };
}
```

```ts
// src/features/game/view/play-machine.ts

/**
 * The four phases of a match session.
 *
 * The owner's architecture named five; `DRAFT_SELECTION` is the round-based Draft Room,
 * which is TASK-1823 / sub-project C. B1 mounts the existing draft hub as `setup`.
 */
export type PlayPhase = "setup" | "preview" | "live" | "summary";

export interface PlayState {
  phase: PlayPhase;
  /** Drawn when the squad is confirmed, so the whole match replays from it. */
  seed: number | null;
}

export type PlayAction =
  | { type: "confirmSquad"; seed: number }
  | { type: "kickOff" }
  | { type: "fullTime" }
  | { type: "newMatch" };

export const createPlayState = (phase: PlayPhase = "setup"): PlayState => ({ phase, seed: null });

/**
 * ⚠️ Transitions that do not belong to the current phase are IGNORED, not applied. A
 * linear flow that ignores by default cannot be driven into a state the UI has no
 * rendering for — which is the reason this is a machine rather than a pile of booleans.
 */
export function playReducer(state: PlayState, action: PlayAction): PlayState {
  switch (action.type) {
    case "confirmSquad":
      return state.phase === "setup" ? { phase: "preview", seed: action.seed } : state;
    case "kickOff":
      return state.phase === "preview" ? { ...state, phase: "live" } : state;
    case "fullTime":
      return state.phase === "live" ? { ...state, phase: "summary" } : state;
    case "newMatch":
      return state.phase === "summary" ? createPlayState("setup") : state;
  }
}
```

```tsx
 * ⚠️ Distinct from `model.lastMinute`. During a live match the model is PARTIAL, so its
 * last minute is only "as far as the engine has been driven" — treating that as full
 * time would announce the result at every segment boundary.
 */
holdAt?: number;
```

Gate the playback tick on it, and gate the full-time rendering on `holdAt == null`.
